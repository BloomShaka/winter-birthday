        // Create falling bread with Fortnite effects
        function createBread() {
            const bread = document.createElement('div');
            bread.className = 'bread-rain';
            bread.innerHTML = '🍞';
            bread.style.left = Math.random() * 100 + '%';
            bread.style.animationDuration = (Math.random() * 3 + 5) + 's';
            bread.style.animationDelay = Math.random() * 5 + 's';
            document.body.appendChild(bread);

            setTimeout(() => bread.remove(), 13000);
        }

        // Generate bread rain
        for (let i = 0; i < 15; i++) {
            createBread();
        }

        setInterval(createBread, 1000);

        function celebrate() {
            const celebration = document.getElementById('celebration');
            const baldieWifey = document.querySelector('.baldie-wifey');
            
            // Show BALDIE WIFEY
            baldieWifey.style.display = 'block';
            
            celebration.style.display = 'block';
            celebration.innerHTML = '🏆 VICTORY ROYALE! 🎂<br>WINTER WINS!';
            
            // Create Fortnite-style confetti
            const emojis = ['🍞', '🖤', '🎮', '🎂', '✨', '⭐', '👑', '🏆'];
            for (let i = 0; i < 40; i++) {
                setTimeout(() => {
                    const confetti = document.createElement('div');
                    confetti.innerHTML = emojis[Math.floor(Math.random() * emojis.length)];
                    confetti.style.position = 'fixed';
                    confetti.style.fontSize = '2.5em';
                    confetti.style.left = Math.random() * 100 + '%';
                    confetti.style.top = '50%';
                    confetti.style.animation = 'fall 3s linear';
                    confetti.style.zIndex = '9999';
                    confetti.style.filter = 'drop-shadow(0 0 10px rgba(138, 43, 226, 0.8))';
                    document.body.appendChild(confetti);
                    setTimeout(() => confetti.remove(), 3000);
                }, i * 50);
            }

            setTimeout(() => {
                celebration.style.display = 'none';
                baldieWifey.style.display = 'none';
            }, 5000);
        }
    </script>
